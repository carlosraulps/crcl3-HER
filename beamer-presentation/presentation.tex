\begin{frame}{Motivation: 2D Magnetic Catalysts for Green \texorpdfstring{$\text{H}_2$}{H2}}
  \begin{columns}[c]
    \begin{column}{0.50\textwidth}
      \begin{block}{Acidic Hydrogen Evolution Reaction (HER)}
        \scriptsize
        Cathodic reaction: $2\text{H}^+ + 2e^- \longrightarrow \text{H}_2, \quad \eta_{\text{HER}} = |\Delta G_{\text{H}^*}|/e$
        \begin{itemize}\setlength{\itemsep}{1pt}
          \item \textbf{Sabatier Ideal}: Near-thermoneutral binding ($\Delta G_{\text{H}^*} \approx 0$).
          \item \textbf{Benchmark Pt (111)}: $\Delta G_{\text{H}^*} = -0.090$~eV (scarce, costly).
        \end{itemize}
      \end{block}
      \vspace{0.1em}
      \begin{exampleblock}{Why Monolayer $\text{CrCl}_3$?}
        \scriptsize
        \begin{itemize}\setlength{\itemsep}{1pt}
          \item \textbf{2D Magnetic Semiconductor}: Exfoliable vdW crystal with intrinsic FM order ($\mu_{\text{Cr}} \approx 3\,\mu_{\text{B}}$) offering a spin degree of freedom.
          \item \textbf{Active-Site Dilemma}: Saturated Cl basal plane is chemically inert $\implies$ Single TM atom engineering creates localized active centers!
        \end{itemize}
      \end{exampleblock}
    \end{column}

    \begin{column}{0.48\textwidth}

      \vspace{0.2em}

      {\scriptsize \textcolor{nordFour}{\textbf{TOC Concept}:} Single-atom functionalization \& coordination engineering.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Computational Methodology \& Energy Partitioning}
  \begin{columns}[c]
    \begin{column}{0.50\textwidth}
      \begin{block}{Spin-Polarized DFT \& AIMD Dynamics}
        \scriptsize
        \begin{itemize}\setlength{\itemsep}{1.2pt}
          \item \textbf{Code \& PAW}: VASP, PAW, GGA-PBE.
          \item \textbf{Supercell}: $2\times 2\times 1$ ($\text{Cr}_8\text{Cl}_{24}$), 15~\AA{} vacuum, $E_{\text{cut}} = 400$~eV, $5\times 5\times 1$ $\Gamma$-mesh, residual forces $< 0.025$~eV/\AA.
          \item \textbf{Hubbard $U$ Benchmark}: Dudarev scheme ($U^* = 3.29$~eV) reproduces experimental optical gap ($2.56$~eV) and linear response ($U_{\text{LR}} = 3.32$~eV) + Grimme DFT-D3(BJ).
          \item \textbf{AIMD 300~K}: 5~ps NVT (Nosé-Hoover, 1~fs step) for finite-temperature kinetic trapping and structural stability.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Computational Hydrogen Electrode (CHE)}
        \scriptsize
        \[
          \Delta G_{\text{H}^*} = E_{\text{surf+H}} - E_{\text{surf}} - \tfrac{1}{2}E_{\text{H}_2} + (\Delta E_{\text{ZPE}} - T\Delta S)
        \]
        Full phononic thermochemistry via explicit Hessians (\texttt{IBRION=5}).
      \end{exampleblock}
      \vspace{0.1em}
      \begin{alertblock}{Distortion-Interaction Decomposition}
        \scriptsize
        Decouples intrinsic chemical binding from substrate distortion:
        \[
          \Delta E_{\text{ads}}^{\text{H}} = E_{\text{int}}^{\text{H}} + E_{\text{def}}
        \]
        \begin{itemize}\setlength{\itemsep}{0.6pt}
          \item $E_{\text{def}} = E_{\text{surf}}^{\text{def}} - E_{\text{surf}} \ge 0$ (lattice distortion strain).
          \item $E_{\text{int}}^{\text{H}} = E_{\text{surf+H}} - E_{\text{surf}}^{\text{def}} - E_{\text{H}} < 0$ (intrinsic affinity).
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

        \scriptsize
        \begin{itemize}\setlength{\itemsep}{0.5pt}
          \item Honeycomb network: $a = 12.10$~\AA{} ($2\times 2$), thickness $h = 2.67$~\AA{}, edge-sharing $\text{CrCl}_6$ ($d = 2.34$~\AA{}).
          \item FM ground state: $M_{\text{tot}} = 24.0\,\mu_{\text{B}}$ ($3\,\mu_{\text{B}}/\text{f.u.}$, $\mu_{\text{Cr}} \approx 2.94\,\mu_{\text{B}}$), direct gap $E_g = 1.50 \to 2.56$~eV (PBE+$U$).
        \end{itemize}
      \end{block}
      \vspace{-0.1em}
      \begin{alertblock}{Basal Plane Catalytic Inertness}
        \scriptsize
        \begin{itemize}\setlength{\itemsep}{0.5pt}
          \item $S_1$ (Top-Cl): Spontaneous H repulsion ($d_{\text{H-surf}} = 3.42$~\AA{}).
          \item $S_2$ (Hollow): Geometrically unstable site.
          \item $S_3$ (Top-Cr): Metastable contact, but severely endergonic: \textcolor{nordEleven}{\textbf{$\Delta G_{\text{H}^*} = +2.44\text{ eV}$}} ($\eta = 2.44$~V).
          \item Pristine basal plane is completely inert for HER!
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

      \begin{block}{Crystal Orbital Hamilton Population (-COHP)}
        \scriptsize
        $\text{ICOHP}_{\text{TM-Cl}} = \int_{-\infty}^{E_{\text{F}}} \text{COHP}(E)\,dE$ (covalent TM-$3d$/Cl-$3p$):
        \begin{itemize}\setlength{\itemsep}{0pt}
          \item Bonding manifold spans $-6.8$ to $-2.5$~eV below $E_{\text{F}}$.
          \item \textbf{Surface (3-fold)}: Strong individual bonds ($\text{ICOHP}_{\text{avg}} = -2.23$~eV for Co, $-2.28$~eV for Fe).
          \item \textbf{Embedded (6-fold)}: Weaker individual bonds ($\text{ICOHP}_{\text{avg}} = -1.87$~eV), but higher cumulative bonding ($\text{ICOHP}_{\text{cum}} \approx -11.4$~eV).
        \end{itemize}
      \end{block}
      \vspace{-0.2em}
      \begin{alertblock}{The Coordination Trade-Off Principle}
        \scriptsize
        High coordination stability does not imply catalysis! Deep 6-fold coordination maximizes host integration at the cost of chemical passivity toward reactants.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}
